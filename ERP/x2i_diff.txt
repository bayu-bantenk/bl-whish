+    def validate_runtime_environment(self) -> None:
+        """X2 — APP_ENV deployment contract.
 
+        Explicit check, never run at import time. Callers must invoke it
+        before validate_production_secrets() and before any database engine
+        is built: main.py lifespan, DatabaseFactory.initialize(cfg) and
+        alembic/env.py.
+
+        * ERP_CONTAINER (raw os.environ, exact values only, no whitespace
+          stripping or case folding): absent -> not a container; "1" ->
+          container; "0" -> explicit operator opt-out; anything else,
+          including "" -> refuse.
+        * R1: container marker active -> APP_ENV must be explicitly provided
+          (init kwarg, environment variable or .env file, i.e. present in
+          model_fields_set). The DEVELOPMENT default does not count.
+        * R2: ERP_DEPLOYMENT_TARGET set -> APP_ENV must equal it.
+
+        Error messages name variables and enum values only, never secrets.
+        """
+        marker = os.environ.get(CONTAINER_MARKER_ENV)
+        if marker is None or marker == "0":
+            in_container = False
+        elif marker == "1":
+            in_container = True
+        else:
+            raise RuntimeError(
+                f"Refusing to start: {CONTAINER_MARKER_ENV} must be exactly '1' (container) "
+                f"or '0' (explicit opt-out) when set; got an unsupported value."
+            )
+
+        if in_container and "APP_ENV" not in self.model_fields_set:
+            raise RuntimeError(
+                f"Refusing to start inside a container ({CONTAINER_MARKER_ENV}=1) without an "
+                f"explicit APP_ENV. Set APP_ENV to one of: "
+                f"{', '.join(e.value for e in Environment)}."
+            )
+
+        target = self.ERP_DEPLOYMENT_TARGET
+        if target is not None and self.APP_ENV.value != target:
+            raise RuntimeError(
+                f"Refusing to start: ERP_DEPLOYMENT_TARGET={target!r} does not match "
+                f"APP_ENV={self.APP_ENV.value!r}."
+            )
